function triggerSearch(updateUrl = true) { const query = document.getElementById('search-input').value.trim(); if (!query) return; if (updateUrl) { const newUrl = window.location.protocol + "//" + window.location.host + window.location.pathname + '?q=' + encodeURIComponent(query); window.history.pushState({ path: newUrl }, '', newUrl); } const container = document.getElementById('results-container'); container.innerHTML = '
מנתח נתונים מוצפנים...
'; // תיקון קריטי: הפיכת השאילתה לאותיות קטנות לפני השליחה לשרת כדי להתאים למאגר const lowercaseQuery = query.toLowerCase(); fetch(`/api/search?q=${encodeURIComponent(lowercaseQuery)}`) .then(res => res.json()) .then(data => { container.innerHTML = ''; if (!data.results || data.results.length === 0) { container.innerHTML = '
🏜️ סופת החול לא מצאה התאמות לביטוי המבוקש.
'; return; } data.results.forEach(item => { const div = document.createElement('div'); div.className = 'result-item'; let words = item.content.split(/\s+/); let computedTitle = words.slice(0, 8).join(" ") + "..."; if (!item.content || item.content.trim() === "") { computedTitle = item.url; } const secureTitle = escapeAndHighlight(computedTitle, query); const secureSnippet = escapeAndHighlight(item.snippet || "", query); div.innerHTML = ` ${secureTitle}
🔗 ${item.url} • מופעים בדף: ${item.score}
${secureSnippet}
`; container.appendChild(div); }); }) .catch(err => { container.innerHTML = '
❌ שגיאה בתקשורת מול השרת.
'; console.error(err); }); }
